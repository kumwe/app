    /**
     * Materialize only the already-authorized projected rows through the admitted native reporting profile.
     *
     * The request-local plan binds the immutable report, its column schema and this execution's limits.
     * No compiled artifact outlives the operation, and native diagnostics never become disclosed cells.
     *
     * @param   ReportDefinition                          $report    Trusted immutable report contribution.
     * @param   list<array<string, bool|int|string|null>>  $rows      Policy-filtered projected records.
     * @param   int                                       $rowLimit  Bound selected for this report or export.
     *
     * @return  list<array<string, bool|int|string|null>>  Native materialization after transport validation.
     *
     * @throws  ReportUnavailable  When the admitted runtime refuses the plan or returns unusable output.
     *
     * @since   2.0.0
     */
    private function materialize(ReportDefinition $report, array $rows, int $rowLimit): array
    {
        $contract = null;
        foreach ($this->compatibility->capabilities->contracts() as $candidate) {
            if ($candidate->owner === 'kumwe/reporting' && $candidate->profile === 'report-materialization-draft/1') {
                $contract = $candidate;
                break;
            }
        }
        if ($contract === null) {
            throw new ReportUnavailable('The report runtime is unavailable.');
        }

        try {
            $definition = $report->toArray();
            $source = array_intersect_key(
                $definition,
                array_fill_keys(['columns', 'groups', 'aggregates', 'formulas', 'sorts'], true),
            );
            $limits = new ExecutionLimits(maxDocuments: 1, maxFindings: 1);
            $program = new ProgramEnvelope($contract, '1.0.0', json_encode($source, JSON_THROW_ON_ERROR));
            $plan = new PlanIdentity(
                $contract,
                $program->programVersion,
                $program->digest(),
                'report-' . $report->version . '-' . $report->checksum(),
                $report->checksum(),
                hash('sha256', json_encode($definition['columns'], JSON_THROW_ON_ERROR)),
                hash('sha256', json_encode([
                    'limits' => $limits->toArray(),
                    'row_limit' => $rowLimit,
                ], JSON_THROW_ON_ERROR)),
                $this->compatibility->capabilities,
            );
            $input = new DocumentInput('report', $contract, json_encode([
                'fields' => ['rows' => $rows],
                'lines' => (object) [],
            ], JSON_THROW_ON_ERROR));
            $compiled = $this->native->compile($program, $plan, $limits);
            try {
                $results = $this->native->execute($compiled, new DocumentBatch([$input], $limits), $limits)->results();
            } finally {
                $this->native->release($compiled);
            }
            if (count($results) !== 1 || $results[0]->findings() !== []) {
                throw new ReportUnavailable('The report runtime returned unusable output.');
            }
            $output = json_decode($results[0]->bytes, true, 64, JSON_THROW_ON_ERROR);
        } catch (ExecutionRefused | JsonException $exception) {
            throw new ReportUnavailable('The report is unavailable.', previous: $exception);
        }

        if (
            !is_array($output)
            || count($output) !== 2
            || !isset($output['rows'], $output['findings'])
            || $output['findings'] !== []
            || !is_array($output['rows'])
            || !array_is_list($output['rows'])
        ) {
            throw new ReportUnavailable('The report runtime returned unusable output.');
        }
        foreach ($output['rows'] as $row) {
            if (!is_array($row) || ($row !== [] && array_is_list($row))) {
                throw new ReportUnavailable('The report runtime returned an invalid row.');
            }
            foreach ($row as $alias => $value) {
                if (
                    !is_string($alias)
                    || (!is_bool($value) && !is_int($value) && !is_string($value) && $value !== null)
                ) {
                    throw new ReportUnavailable('The report runtime returned an invalid cell.');
                }
            }
        }

        return $output['rows'];
    }
