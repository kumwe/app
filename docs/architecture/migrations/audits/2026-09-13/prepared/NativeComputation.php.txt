<?php

declare(strict_types=1);

namespace Kumwe\App\Tests\Support;

use Kumwe\App\Kernel\NativeComputationFactory;
use Kumwe\App\Shared\Infrastructure\Configuration\Environment;
use Kumwe\CanonicalJson\CanonicalEncoder;
use Kumwe\Computation\NativeAdapter;
use Kumwe\Computation\NativeCompatibility;
use Kumwe\Engine\Runtime;

/**
 * Construct real native collaborators from the test deployment's independently recorded runtime tuple.
 *
 * Host tests use the production admission boundary and require a provisioned extension plus an
 * independently recorded deployment tuple. Computations execute through that admitted native runtime.
 *
 * @since  2.0.0
 */
final class NativeComputation
{
    /**
     * Load the exact expected build and semantic contracts selected by test provisioning.
     *
     * @return  NativeCompatibility  Independently recorded deployment identity.
     *
     * @since   2.0.0
     */
    public static function compatibility(): NativeCompatibility
    {
        return (new NativeComputationFactory())->compatibility(Environment::fromGlobals());
    }

    /**
     * Construct the public compiler and executor against the actual installed extension.
     *
     * @return  NativeAdapter  Admitted native compiler and executor with request-local plan ownership.
     *
     * @since   2.0.0
     */
    public static function adapter(): NativeAdapter
    {
        return new NativeAdapter(new Runtime(), self::compatibility());
    }

    /**
     * Supply the package encoder already used by the production composition root.
     *
     * @return  CanonicalEncoder  Actual native canonical encoder.
     *
     * @since   2.0.0
     */
    public static function encoder(): CanonicalEncoder
    {
        return (new NativeComputationFactory())->create(Environment::fromGlobals());
    }
}
